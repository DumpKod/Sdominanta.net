// Редактор витрины хранителя. Отдаётся как /k/editor.js только при действующей сессии (__Host-session):
// без сессии этого кода в браузере нет. Работает поверх движка /vitrina.js (событие vitrina:ready), модель — E.M.
// Холст как в графических редакторах: пробел + перетаскивание или средняя кнопка — сдвиг, колёсико — сдвиг,
// Ctrl/⌘ + колёсико — масштаб, Shift+1 — вся сцена, Shift+2 — к выделенному, Ctrl+Z / Ctrl+Shift+Z — отмена и повтор.
// Инструменты: смотреть (как гость), выбор (клик по объекту или внутри его рамки, Shift — добавить, рамка, перетаскивание;
//   у рамки выделенного — ручки: углы и стороны растягивают (Shift — пропорционально, Alt — от центра), кружок сверху поворачивает,
//   Shift — шагами по 15°),
// точки (перетащить точку жилы, клик по жиле — новая точка, двойной клик или Delete — убрать точку),
// кисть (мазок в выбранный объект), ластик (клик по жиле), область (кистью по узлам: мазок закрашивает узлы, Alt — стирает;
//   закрашенное обводится контуром — это область окна объекта).
// Панель свойств выделенного объекта: подпись, адрес, слой, параметры жил (tubeR, толщина, резонанс, такты…), окно и его рама.
// Кнопки «Сохранить черновик», «Опубликовать», «Откатить» — POST в /k/vitrina/* с CSRF-токеном сессии.
(() => {
  'use strict';
  const csrf = (document.querySelector('meta[name="k-csrf"]') || {}).content || '';
  const start = (api) => {
    if (start.done) return;
    start.done = true;
    init(api);
  };
  if (window.vitrinaEngine) start(window.vitrinaEngine);
  document.addEventListener('vitrina:ready', (e) => start(e.detail));

  function init(E) {
    const M = E.M, cv = E.cv;
    const version = cv.dataset.version || '';
    const el = (tag, cls, text) => {
      const n = document.createElement(tag);
      if (cls) n.className = cls;
      if (text) n.textContent = text;
      return n;
    };
    const button = (parent, name, fn, cls, title) => {
      const b = el('button', 'kp-b' + (cls ? ' ' + cls : ''), name);
      b.type = 'button';
      if (title) b.title = title;
      b.addEventListener('click', fn);
      parent.append(b);
      return b;
    };
    const input = (tag) => el(tag || 'input', 'kp-in');
    const wrap = (inp, text) => {
      const l = el('label', 'kp-f');
      l.append(el('span', '', text), inp);
      return l;
    };
    const section = (title, open) => {
      const d = el('details', 'kp-sec');
      d.open = !!open;
      d.append(el('summary', '', title));
      return d;
    };

    /* ---------- панель ---------- */
    const panel = el('div', 'kp');
    const head = el('p', 'kp-note', version ? `Предпросмотр версии ${version}` : 'Витрина: опубликованная сцена');
    const toolsRow = el('div', 'kp-row');
    const TOOLS = { view: 'Смотреть', select: 'Выбор', points: 'Точки', brush: 'Кисть', erase: 'Ластик', area: 'Область' };
    const btns = {};
    for (const [t, name] of Object.entries(TOOLS)) btns[t] = button(toolsRow, name, () => setTool(t));
    // Размер кисти — в пикселях экрана: мазок получает радиус в мире по текущему масштабу (приблизил — тоньше в мире).
    const brushRow = el('label', 'kp-f');
    const brushOut = el('span', '', '');
    const brushR = el('input', 'kp-range');
    brushR.type = 'range'; brushR.min = '1'; brushR.max = '60'; brushR.step = '0.5'; brushR.value = '4';
    const showBrush = () => { brushOut.textContent = `Кисть: ${brushR.value} px на экране`; };
    brushR.addEventListener('input', showBrush);
    showBrush();
    brushRow.append(brushOut, brushR);
    const brushWorld = () => Math.round((+brushR.value / E.cam.zoom) * 100) / 100;
    let brushAt = null; // курсор кисти на экране
    const histRow = el('div', 'kp-row');
    const bUndo = button(histRow, 'Отменить', undo, '', 'Ctrl+Z');
    const bRedo = button(histRow, 'Повторить', redo, '', 'Ctrl+Shift+Z');
    button(histRow, 'Вся сцена', () => E.fitAll(), '', 'Shift+1');
    button(histRow, 'К выделенному', fitSelection, '', 'Shift+2');
    button(histRow, 'Предпросмотр', () => setPreview(true), '', 'Витрина как у гостя, с несохранёнными правками');
    const pick = el('select', 'kp-sel');
    const objRow = el('div', 'kp-row');

    const props = section('Объект', true);
    const fId = input(), fLabel = input(), fHref = input(), fInner = input(), fSlot = input('select'), fZ = input(), fTitle = input(), fText = input('textarea'), fLinks = input('textarea'), fWidget = input('select'), fRel = input('textarea');
    fRel.rows = 3;
    for (const [v, t] of [['', '— нет —'], ['player', 'плеер'], ['keeper', 'вход хранителя']]) fSlot.append(new Option(t, v));
    const WIDGET_NAMES = { player: 'плеер', keeper: 'вход хранителя', queue: 'очередь (хранителю)', music: 'музыка (хранителю)', vitrina: 'витрина (хранителю)' };
    fWidget.append(new Option('— нет —', ''));
    for (const w of M.WIDGETS) fWidget.append(new Option(WIDGET_NAMES[w] || w, w));
    fZ.type = 'number'; fZ.step = '1'; fZ.min = '-100'; fZ.max = '100';
    fText.rows = 4;
    fLinks.rows = 3;
    props.append(
      wrap(fId, 'id: латиница, цифры, дефис'),
      wrap(fLabel, 'Подпись (видят модели в <nav>)'),
      wrap(fHref, 'Адрес: относительный путь, например /wall'),
      wrap(fSlot, 'Место: плеер или вход хранителя встанут в рамку объекта'),
      wrap(fInner, 'Алеф-нуль: имя пространства внутри (латиница, цифры, дефис), пусто — не вход'),
      wrap(fZ, 'Слой: выше — поверх и первым ловит клик'),
      wrap(fRel, 'Связи узла: по строке «id вес», вес 0…1 (до 10) — при входе внутрь нити сквозь горизонт'),
    );
    const STYLE_NAMES = { tubeR: 'Радиус трубки (tubeR)', thick: 'Толщина ×', resonance: 'Резонанс', tact: 'Такты', flow: 'Течение', wobble: 'Волнистость', fill: 'Заливка жилы', sound: 'Реакция на звук (0 — нет)' };
    const sliders = {};
    const styleBox = section('Параметры жил объекта');
    for (const [k, [lo, hi]] of Object.entries(M.STYLE)) sliders[k] = slider(styleBox, k, STYLE_NAMES[k], lo, hi, 'style');
    button(styleBox, 'tubeR — всем жилам объекта', () => {
      const o = first();
      if (!o) return;
      const r = o.tubeR();
      for (const s of o.strokes) { s.r = r; for (const p of s.pts) p.length = 2; }
      changed(true);
    });
    // Входы объекта, по шагам: 1) выбрать область (основная или ещё одна — у объекта их может быть много);
    // 2) закрасить её кистью «Область» по узлам; 3) что внутри; 4) как взрывается и раскрывается; 5) проверить.
    // Вход срабатывает в пределе приближения над областью: колёсико копит напор, за порогом — взрыв внутрь.
    const winBox = section('Входы: области и окна', true);
    const fDoor = input('select'), dLabel = input(), dHref = input(), dInner = input();
    const winState = el('p', 'kp-note', '');
    const winRow = el('div', 'kp-row');
    button(winRow, 'Закрасить область', () => setTool('area'), '', 'Инструмент «Область»: кистью по узлам, Alt — стереть');
    button(winRow, 'Убрать область', () => {
      const o = first();
      if (!o) return;
      if (doorIx >= 0) { o.doors.splice(doorIx, 1); if (!o.doors.length) o.doors = null; doorIx = -1; }
      else o.area = null;
      changed(true);
      showFields();
    }, '', 'Выбранную в списке: основную — только контур, область-вход — целиком');
    button(winRow, 'Проверить вход', () => {
      const o = first();
      if (!o) return;
      readFields();
      setPreview(true);
      if (!E.openFrameOf(doorIx >= 0 ? `${o.id}.${doorIx + 1}` : o.id)) note.textContent = 'Входу нужны область и содержимое: текст, адрес, виджет или алеф-нуль.';
    }, '', 'Предпросмотр: долететь к области и взорваться внутрь');
    const doorOnly = el('div', '');
    doorOnly.append(
      wrap(dLabel, 'Подпись входа (видят модели в <nav>); пусто — заголовок окна или подпись объекта'),
      wrap(dHref, 'Адрес страницы внутри: относительный путь, например /wall/rules'),
      wrap(dInner, 'Алеф-нуль внутри: имя пространства, пусто — не портал'),
    );
    const mainNote = el('p', 'kp-note', 'У основной области адрес и алеф-нуль — поля объекта выше.');
    winBox.append(
      wrap(fDoor, 'Область объекта: основная или ещё одна (до 16), у каждой своё содержимое'),
      winState, winRow, mainNote, doorOnly,
      wrap(fWidget, 'Виджет: живое управление (разделы хранителя видит только хранитель)'),
      wrap(fTitle, 'Заголовок окна'), wrap(fText, 'Текст окна: по абзацу на строку'), wrap(fLinks, 'Ссылки, по строке «подпись | /путь»'),
      el('p', 'kp-note', 'Порядок содержимого: алеф-нуль, виджет, текст окна, страница по адресу.'),
    );
    const FRAME_NAMES = {
      push: 'Порог напора: сила колёсика до взрыва', depth: 'Глубина провала (пустота континуума)', width: 'Ширина колонки текста, px',
      far: 'Поле под текстом тише', aspect: 'Высота к ширине (старая рамка)', round: 'Форма портала: 0 прямоугольник — 1 овал', ring: 'Кольцо узлов портала, px',
    };
    const frameSliders = {};
    for (const k of ['push', 'depth', 'width', 'far', 'round', 'ring', 'aspect']) {
      const [lo, hi] = M.FRAME[k];
      frameSliders[k] = slider(winBox, k, FRAME_NAMES[k], lo, hi, 'frame');
    }
    let doorIx = -1; // -1 — основная область объекта, иначе номер в o.doors
    /** Что правят поля окна: основная область — сам объект, область-вход — её запись в o.doors. */
    function tgt() {
      const o = first();
      if (!o) return null;
      if (doorIx >= 0 && o.doors && o.doors[doorIx]) return o.doors[doorIx];
      doorIx = -1;
      return o;
    }
    const frameOf = (t, k) => (t && t.frame && typeof t.frame[k] === 'number' ? t.frame[k] : M.FRAME[k][2]);
    function fillDoors(o) {
      fDoor.textContent = '';
      fDoor.append(new Option('Основная область' + (o && o.area ? '' : ' (не закрашена)'), '-1'));
      ((o && o.doors) || []).forEach((d, k) => fDoor.append(new Option(`Область ${k + 2}${d.label ? ' — ' + d.label : ''}${d.area ? '' : ' (не закрашена)'}`, String(k))));
      fDoor.append(new Option('+ новая область', 'new'));
      fDoor.value = String(doorIx);
      fDoor.disabled = !o;
    }
    fDoor.addEventListener('change', () => {
      const o = first();
      if (!o) return;
      if (fDoor.value === 'new') {
        if ((o.doors || []).length >= M.DOORS_MAX) { note.textContent = `У объекта до ${M.DOORS_MAX} областей-входов.`; fDoor.value = String(doorIx); return; }
        o.doors = o.doors || [];
        o.doors.push({});
        doorIx = o.doors.length - 1;
        setTool('area');
        note.textContent = 'Новая область: закрасьте её кистью по узлам, затем задайте, что внутри.';
      } else doorIx = +fDoor.value;
      showFields();
    });
    /** Что раскроется: та же очерёдность, что у движка (алеф-нуль, виджет, окно, страница). */
    function winKind(t) {
      if (!t) return 'Выберите объект.';
      const has = t.area && t.area.length >= 3;
      const what = t.inner ? `алеф-нуль «${t.inner}»` : t.widget ? `виджет «${WIDGET_NAMES[t.widget] || t.widget}»` : t.window ? 'окно с текстом' : t.href && /^\/(?!\/)/.test(t.href) && !/^\/(k|in)(\/|$)/.test(t.href) ? `страница ${t.href}` : null;
      if (!has && !what) return 'Входа нет: закрасьте область и задайте содержимое.';
      if (!has) return `Внутри будет ${what}, но области нет: закрасьте её инструментом «Область».`;
      if (!what) return 'Область есть, содержимого нет: текст, адрес, виджет или алеф-нуль.';
      return `Готово: напор колёсика над областью взорвёт её внутрь — ${what}.`;
    }
    const SPACE_NAMES = { drift: 'Дрейф поля', density: 'Плотность', glow: 'Свечение', haze: 'Дымка (мягкий слой под линиями)', sound: 'Реакция поля на звук (0 — нет)', reach: 'Радиус музыки вокруг плеера (доля сцены)', horizon: 'Горизонт уровня выше (0 — чистый холст)' };
    const spaceBox = section('Пространство сцены');
    const spaceSliders = {};
    for (const [k, [lo, hi]] of Object.entries(M.SPACE)) spaceSliders[k] = slider(spaceBox, k, SPACE_NAMES[k], lo, hi, 'space');

    const sceneRow = el('div', 'kp-row');
    const note = el('p', 'kp-note', 'Правки живут на этой странице, пока не сохранены черновиком.');
    const keys = el('p', 'kp-note', 'Пробел + мышь или средняя кнопка — сдвиг · Ctrl + колёсико — масштаб · Shift+1 / Shift+2 · Ctrl+Z');
    const box = el('textarea', 'kp-json');
    box.spellcheck = false;
    box.hidden = true;
    panel.append(head, toolsRow, brushRow, histRow, pick, objRow, props, styleBox, winBox, spaceBox, sceneRow, note, keys, box);
    document.body.append(panel);

    // Предпросмотр: витрина как у гостя — без панели, рамок, мини-карты и места хранителя; правка не теряется.
    let preview = false;
    const back = el('button', 'kp-b kp-back', 'Вернуться к правке');
    back.type = 'button';
    back.hidden = true;
    document.body.append(back);
    function setPreview(on) {
      preview = on;
      if (on) { readFields(); setTool('view'); }
      panel.hidden = on;
      back.hidden = !on;
      document.documentElement.classList.toggle('kp-preview', on);
      E.rebuild(1);
    }
    back.addEventListener('click', () => setPreview(false));
    for (const n of panel.querySelectorAll('input, textarea, select')) n.addEventListener('keydown', (e) => e.stopPropagation());

    function slider(parent, k, name, lo, hi, kind) {
      const row = el('label', 'kp-f kp-sl');
      const top = el('span', '', name);
      const out = el('output', 'kp-out');
      const reset = el('button', 'kp-x', '×');
      reset.type = 'button';
      reset.title = 'по умолчанию';
      top.append(out, reset);
      const r = el('input', 'kp-range');
      r.type = 'range'; r.min = String(lo); r.max = String(hi); r.step = String((hi - lo) / 100);
      row.append(top, r);
      parent.append(row);
      const target = () => (kind === 'space' ? E.scene : kind === 'frame' ? tgt() : first());
      r.addEventListener('input', () => {
        const t = target();
        if (!t) return;
        const key = kind;
        t[key] = t[key] || {};
        t[key][k] = Math.round(+r.value * 1000) / 1000;
        out.textContent = String(t[key][k]);
        reset.hidden = false;
        E.rebuild(4);
      });
      r.addEventListener('change', () => changed(false));
      reset.addEventListener('click', (e) => {
        e.preventDefault();
        const t = target(), key = kind;
        if (!t || !t[key]) return;
        delete t[key][k];
        if (!Object.keys(t[key]).length) t[key] = null;
        showFields();
        changed(true);
      });
      return { r, out, reset, k, kind };
    }

    /* ---------- состояние: выделение (по id) и история ---------- */
    let tool = 'view';
    let selIds = [];
    const hist = new M.History();
    const snap = () => JSON.stringify(E.exportScene());
    hist.reset(snap());
    // Переход в другое пространство (алеф-нуль): своя сцена — своя история, выделение сбрасывается.
    document.addEventListener('vitrina:space', () => {
      selIds = [];
      hist.reset(snap());
      fillPick();
      showFields();
      note.textContent = `Пространство: ${E.space}.`;
    });
    const byId = (id) => E.objects.find((o) => o.id === id) || null;
    const selected = () => selIds.map(byId).filter(Boolean);
    const first = () => selected()[0] || null;
    function select(ids) {
      const was = selIds[0];
      selIds = ids.filter((id, i) => ids.indexOf(id) === i && byId(id));
      if (selIds[0] !== was) doorIx = -1;
      fillPick();
      showFields();
      E.rebuild(1);
    }
    /** Правка закончена: записать в историю; rebuild — пересобрать жилы движка. */
    function changed(rebuild) {
      if (rebuild) E.rebuild(4);
      hist.commit(snap());
      updateHist();
    }
    function updateHist() {
      bUndo.disabled = !hist.canUndo;
      bRedo.disabled = !hist.canRedo;
    }
    function restore(state) {
      if (!state) return;
      E.load(JSON.parse(state));
      selIds = selIds.filter(byId);
      fillPick();
      showFields();
      updateHist();
    }
    function undo() { restore(hist.undo()); }
    function redo() { restore(hist.redo()); }

    function setTool(t) {
      tool = t;
      E.setTool(t);
      for (const [k, b] of Object.entries(btns)) b.classList.toggle('on', k === t);
      if (t !== 'area') paint = null;
      point = null;
    }
    function fitSelection() {
      const b = selected().reduce((a, o) => M.union(a, o.bounds()), { ...M.EMPTY });
      if (!M.isEmpty(b)) E.flyTo(M.expand(b, Math.max(b.x1 - b.x0, b.y1 - b.y0) * 0.15));
    }

    function fillPick() {
      pick.textContent = '';
      pick.append(new Option('— ничего не выбрано —', ''));
      E.objects.forEach((o) => pick.append(new Option(`${o.id} — ${o.label}`, o.id)));
      pick.value = selIds[0] || '';
    }
    function showFields() {
      const o = first();
      for (const f of [fId, fLabel, fHref, fInner, fSlot, fWidget, fZ, fTitle, fText, fLinks, dLabel, dHref, dInner]) f.disabled = !o;
      const t = tgt();
      fillDoors(o);
      doorOnly.hidden = doorIx < 0; mainNote.hidden = doorIx >= 0;
      dLabel.value = (doorIx >= 0 && t.label) || '';
      dHref.value = (doorIx >= 0 && t.href) || '';
      dInner.value = (doorIx >= 0 && t.inner) || '';
      fWidget.value = (t && t.widget) || '';
      fRel.value = ((o && o.links) || []).map((l) => `${l.to} ${l.w ?? 0.5}`).join('\n');
      fRel.disabled = !o;
      for (const s of [...Object.values(sliders), ...Object.values(frameSliders)]) s.r.disabled = !o;
      fId.value = o ? o.id : '';
      fLabel.value = o ? o.label : '';
      fHref.value = (o && o.href) || '';
      fInner.value = (o && o.inner) || '';
      fSlot.value = (o && o.slot) || '';
      fZ.value = o ? String(o.z || 0) : '';
      fTitle.value = (t && t.window && t.window.title) || '';
      fText.value = (t && t.window && t.window.text) || '';
      fLinks.value = ((t && t.window && t.window.links) || []).map((l) => `${l.label} | ${l.href}`).join('\n');
      for (const s of Object.values(sliders)) {
        const own = !!(o && o.style && typeof o.style[s.k] === 'number');
        const v = o ? (s.k === 'tubeR' ? o.tubeR() : o.get(s.k)) : M.STYLE[s.k][2];
        s.r.value = String(v);
        s.out.textContent = String(Math.round(v * 1000) / 1000) + (own ? '' : ' (общий)');
        s.reset.hidden = !own;
      }
      for (const s of Object.values(frameSliders)) {
        const own = !!(t && t.frame && typeof t.frame[s.k] === 'number');
        const v = frameOf(t, s.k);
        s.r.value = String(v);
        s.out.textContent = String(Math.round(v * 1000) / 1000) + (own ? '' : ' (общий)');
        s.reset.hidden = !own;
      }
      winState.textContent = winKind(t);
      for (const s of Object.values(spaceSliders)) {
        const own = !!(E.scene.space && typeof E.scene.space[s.k] === 'number');
        const v = E.scene.get(s.k);
        s.r.value = String(v);
        s.out.textContent = String(v) + (own ? '' : ' (общий)');
        s.reset.hidden = !own;
      }
      props.querySelector('summary').textContent = selIds.length > 1 ? `Объект (выбрано ${selIds.length}, свойства первого)` : 'Объект';
    }
    function readFields() {
      const o = first();
      if (!o) return;
      o.label = fLabel.value.trim() || o.id;
      o.href = fHref.value.trim() || null;
      o.inner = fInner.value.trim().toLowerCase() || null;
      o.slot = fSlot.value || null;
      const t = tgt();

    let drag = null, raw = [], paint = null, band = null, point = null, hoverPt = null;

    /** Ближайшая точка жилы выделенных объектов в пределах d пикселей экрана. */
    function nearPoint(x, y, d) {
      let best = null;
      for (const o of selected())
        o.strokes.forEach((s, si) => s.pts.forEach((p, pi) => {
          const [sx, sy] = E.toScreen(p[0], p[1]), dd = Math.hypot(sx - x, sy - y);
          if (dd < d && (!best || dd < best.d)) best = { o, si, pi, d: dd };
        }));
      return best;
    }
    /** Ближайший отрезок жилы выделенных объектов: куда вставить новую точку. */
    function nearSegment(x, y, d) {
      let best = null;
      for (const o of selected())
        o.strokes.forEach((s, si) => {
          const pts = s.closed ? [...s.pts, s.pts[0]] : s.pts;
          for (let i = 0; i < pts.length - 1; i++) {
            const [ax, ay] = E.toScreen(pts[i][0], pts[i][1]), [bx, by] = E.toScreen(pts[i + 1][0], pts[i + 1][1]);
            const ex = bx - ax, ey = by - ay, t = Math.max(0, Math.min(1, ((x - ax) * ex + (y - ay) * ey) / (ex * ex + ey * ey || 1e-9)));
            const dd = Math.hypot(x - ax - ex * t, y - ay - ey * t);
            if (dd < d && (!best || dd < best.d)) best = { o, si, at: i + 1, d: dd };
          }
        });
      return best;
    }
    /** Объект под точкой для выбора: по жилам и областям, иначе — самый маленький, чья рамка содержит точку. */
    function pickAt(x, y) {
      const i = E.hit(x, y);
      if (i >= 0) return i;
      const [wx, wy] = E.toWorld(x, y);
      let best = -1, area = Infinity;
      E.objects.forEach((o, j) => {
        const b = o.bounds(), a = (b.x1 - b.x0) * (b.y1 - b.y0);
        if (wx >= b.x0 && wx <= b.x1 && wy >= b.y0 && wy <= b.y1 && a < area) { best = j; area = a; }
      });
      return best;
    }
    /* ---------- ручки рамки выделенного: растяжение и поворот ---------- */
    const HANDLE = 6;
    function selBox() {
      const b = selected().reduce((a, o) => M.union(a, o.bounds()), { ...M.EMPTY });
      if (M.isEmpty(b)) return null;
      const [x0, y0] = E.toScreen(b.x0, b.y0), [x1, y1] = E.toScreen(b.x1, b.y1);
      return { b, x0: x0 - 3, y0: y0 - 3, x1: x1 + 3, y1: y1 + 3 };
    }
    /** Ручки в пикселях экрана: fx, fy — доля рамки (0, ½, 1); rot — поворот. */
    function handles() {
      const s = selBox();
      if (!s) return [];
      const out = [];
      for (const fy of [0, 0.5, 1]) for (const fx of [0, 0.5, 1]) {
        if (fx === 0.5 && fy === 0.5) continue;
        out.push({ fx, fy, x: s.x0 + (s.x1 - s.x0) * fx, y: s.y0 + (s.y1 - s.y0) * fy });
      }
      out.push({ rot: true, x: (s.x0 + s.x1) / 2, y: s.y0 - 22 });
      return out;
    }
    function handleAt(x, y) {
      return handles().find((h) => Math.abs(h.x - x) <= HANDLE + 2 && Math.abs(h.y - y) <= HANDLE + 2) || null;
    }
    /** Начало правки ручкой: снимок жил и областей, чтобы каждый шаг считался от исходного вида без накопления округлений. */
    function startHandle(h, wx, wy) {
      const b = selected().reduce((a, o) => M.union(a, o.bounds()), { ...M.EMPTY });
      const orig = selected().map((o) => ({ o, strokes: JSON.stringify(o.strokes), area: o.area ? JSON.stringify(o.area) : null, doors: o.doors ? JSON.stringify(o.doors) : null }));
      drag = { kind: h.rot ? 'rotate' : 'stretch', h, b, orig, start: [wx, wy], moved: false };
    }
    function applyHandle(e, wx, wy) {
      const { h, b, orig } = drag;
      for (const r of orig) { r.o.strokes = JSON.parse(r.strokes); r.o.area = r.area ? JSON.parse(r.area) : null; r.o.doors = r.doors ? JSON.parse(r.doors) : null; }
      const cx = (b.x0 + b.x1) / 2, cy = (b.y0 + b.y1) / 2;
      if (drag.kind === 'rotate') {
        let a = Math.atan2(wy - cy, wx - cx) - Math.atan2(drag.start[1] - cy, drag.start[0] - cx);
        if (e.shiftKey) a = Math.round(a / (Math.PI / 12)) * (Math.PI / 12);
        for (const r of orig) r.o.rotate(a, cx, cy);
        drag.angle = a;
      } else {
        // опора — противоположная сторона (Alt — центр рамки)
        const hx = b.x0 + (b.x1 - b.x0) * h.fx, hy = b.y0 + (b.y1 - b.y0) * h.fy;
        const px = e.altKey ? cx : b.x0 + (b.x1 - b.x0) * (1 - h.fx), py = e.altKey ? cy : b.y0 + (b.y1 - b.y0) * (1 - h.fy);
        const lim = (v) => (Math.abs(v) < 0.02 ? (v < 0 ? -0.02 : 0.02) : v);
        let sx = h.fx === 0.5 ? 1 : lim((wx - px) / (hx - px || 1e-9));
        let sy = h.fy === 0.5 ? 1 : lim((wy - py) / (hy - py || 1e-9));
        if (e.shiftKey) {
          const k = h.fx === 0.5 ? sy : h.fy === 0.5 ? sx : Math.sign(sx) * Math.max(Math.abs(sx), Math.abs(sy));
          sx = Math.sign(sx || 1) * Math.abs(k); sy = Math.sign(sy || 1) * Math.abs(k);
        }
        for (const r of orig) r.o.stretch(sx, sy, px, py);
        drag.k = [sx, sy];
      }
      drag.moved = true;
      E.rebuild(2);
    }

    function deletePoint(pt) {
      const s = pt.o.strokes[pt.si];
      s.pts.splice(pt.pi, 1);
      if (!s.pts.length) pt.o.strokes.splice(pt.si, 1);
      point = null;
      changed(true);
    }

    E.setInput({

          changed(true);
        } else if (tool === 'area') {
          if (!first()) { const i = pickAt(x, y); if (i >= 0) select([E.objects[i].id]); }
          paint = { pts: [[wx, wy]], r: brushWorld(), sub: e.altKey };
        }
      },
      move(e) {
        const x = e.clientX, y = e.clientY, [wx, wy] = E.toWorld(x, y);
        if (tool === 'points' && !drag) hoverPt = nearPoint(x, y, 8);
        if (tool === 'brush' || tool === 'area') brushAt = [x, y]; // кадр и так перерисовывается: движок в движении
        if (paint) {
          const l = paint.pts[paint.pts.length - 1];
          if (Math.hypot(wx - l[0], wy - l[1]) > paint.r * 0.4) paint.pts.push([wx, wy]);
          return;
        }
        if (!drag && !band && !raw.length) return;
        if (drag && drag.kind === 'mini') miniJump(x, y);
        else if (drag && (drag.kind === 'stretch' || drag.kind === 'rotate')) applyHandle(e, wx, wy);
        else if (drag && drag.kind === 'move') {
          const dx = wx - drag.last[0], dy = wy - drag.last[1];
          if (!dx && !dy) return;
          for (const o of selected()) o.translate(dx, dy);
          drag.last = [wx, wy];
          drag.moved = true;
          E.rebuild(2);
        } else if (drag && drag.kind === 'point' && point) {
          const p = point.o.strokes[point.si].pts[point.pi];
          p[0] = r1(wx); p[1] = r1(wy);
          drag.moved = true;
          E.rebuild(2);
        } else if (band) {
          band.x1 = x; band.y1 = y;
        } else if (raw.length) {
          const l = raw[raw.length - 1];
          if (Math.hypot(wx - l[0], wy - l[1]) * E.cam.zoom > 1.5) {
            raw.push([wx, wy]);

          }
        }
      },
      up() {

        drag = null;
        if (band) {
          const [ax, ay] = E.toWorld(band.x0, band.y0), [bx, by] = E.toWorld(band.x1, band.y1);

          raw = [];
          E.setLive(null);
          if (o && !o.strokes.length && !o.area) E.objects.splice(E.objects.indexOf(o), 1);
          changed(true);
        }
      },
      cancel() {
        if (drag && drag.orig) { for (const r of drag.orig) { r.o.strokes = JSON.parse(r.strokes); r.o.area = r.area ? JSON.parse(r.area) : null; r.o.doors = r.doors ? JSON.parse(r.doors) : null; } E.rebuild(2); }
        drag = null; band = null; paint = null;
        if (raw.length) { raw = []; E.setLive(null); }
      },
      dblclick(e) {
        if (tool === 'points') {
          const p = nearPoint(e.clientX, e.clientY, 8);
          if (p) deletePoint(p);

    /** Мазок области закончен: объект — выбранный, иначе тот, чьих точек жил под мазком больше. */
    function finishPaint() {
      const p = paint;
      paint = null;
      let o = first();
      if (!o) {
        let best = 0;
        for (const ob of E.objects) {
          let n = 0;
          for (const s of ob.strokes) for (const q of s.pts) if (p.pts.some((c) => Math.hypot(c[0] - q[0], c[1] - q[1]) < p.r)) n++;
          if (n > best) { best = n; o = ob; }
        }
        if (!o) { note.textContent = 'Выберите объект или ведите кистью области по его жилам.'; return; }
        select([o.id]);
      }
      const t = doorIx >= 0 && o.doors && o.doors[doorIx] ? o.doors[doorIx] : o;
      const area = paintArea(t.area, p.pts, p.r, p.sub);
      t.area = area;
      note.textContent = (doorIx >= 0 ? `Область ${doorIx + 2}: ` : 'Основная область: ') + (area ? `${area.length} точек контура.` : 'стёрта.');
      changed(true);
      showFields();
    }
    function miniJump(x, y) {
      const mr = miniRect();
      E.cam.x = (x - mr.x - miniMap.ox) / miniMap.s;
      E.cam.y = (y - mr.y - miniMap.oy) / miniMap.s;
      E.cameraChanged();
    }

    window.addEventListener('keydown', (e) => {
      if (e.target && /^(INPUT|TEXTAREA|SELECT)$/.test(e.target.tagName)) return;
      const mod = e.ctrlKey || e.metaKey;
      if (mod && (e.key === 'z' || e.key === 'Z' || e.code === 'KeyZ')) { e.preventDefault(); e.shiftKey ? redo() : undo(); return; }
      if (mod && (e.key === 'y' || e.code === 'KeyY')) { e.preventDefault(); redo(); return; }
      if (e.code === 'Digit2' && e.shiftKey) { e.preventDefault(); fitSelection(); return; }
      if (tool === 'view') return;
      if (e.key === 'Escape') { paint = null; band = null; point = null; if (tool === 'select') select([]); }
      if (e.key === 'Delete' || e.key === 'Backspace') {
        e.preventDefault();
        if (tool === 'points' && point) deletePoint(point);
        else if (tool === 'select') deleteSelected();
      }
    });

    /* ---------- поверх кадра: выделение, точки, область, рамка выбора, мини-карта ---------- */
    E.setOverlay((c) => {
      // узлы в областях выделенных объектов светятся (в предпросмотре — нет: как у гостя)
      E.marks = preview || tool === 'view' ? [] : selected().flatMap((o) => o.areas());
      if (preview) return;
      const W = E.W, H = E.H;
      c.lineWidth = 1;
      if (tool !== 'view') {
        c.setLineDash([3, 6]);
        for (const o of selected()) {
          const b = o.bounds();
          if (M.isEmpty(b)) continue;
          const [x0, y0] = E.toScreen(b.x0, b.y0), [x1, y1] = E.toScreen(b.x1, b.y1);
          c.strokeStyle = 'rgba(0, 242, 254, 0.6)';
          c.strokeRect(x0 - 3, y0 - 3, x1 - x0 + 6, y1 - y0 + 6);
          // области объекта: выбранная в списке — ярче, у каждой номер
          const list = [[o.area, -1], ...(o.doors || []).map((d, k) => [d.area, k])];
          for (const [a, k] of list) {
            if (!a || a.length < 3) continue;
            const on = o === first() && k === doorIx;
            c.strokeStyle = on ? 'rgba(250, 204, 21, 0.95)' : 'rgba(250, 204, 21, 0.45)';
            c.fillStyle = on ? 'rgba(250, 204, 21, 0.09)' : 'rgba(250, 204, 21, 0.03)';
            c.beginPath();
            a.forEach(([x, y], j) => { const [sx, sy] = E.toScreen(x, y); j ? c.lineTo(sx, sy) : c.moveTo(sx, sy); });
            c.closePath();
            c.fill();
            c.stroke();
            const [lx, ly] = E.toScreen(a[0][0], a[0][1]);
            c.fillStyle = 'rgba(250, 204, 21, 0.9)';
            c.font = '11px ui-monospace, monospace';
            c.fillText(k < 0 ? '1' : String(k + 2), lx + 4, ly - 4);
          }
        }
        if (tool === 'select' && selIds.length) {
          for (const hd of handles()) {
            c.fillStyle = 'rgba(2, 8, 20, 0.9)';
            c.strokeStyle = 'rgba(165, 243, 252, 0.95)';
            c.beginPath();
            if (hd.rot) c.arc(hd.x, hd.y, HANDLE - 1, 0, Math.PI * 2);
            else c.rect(hd.x - HANDLE / 2 - 1, hd.y - HANDLE / 2 - 1, HANDLE + 2, HANDLE + 2);
            c.fill();
            c.stroke();
          }
          if (drag && drag.kind === 'rotate' && typeof drag.angle === 'number') {
            c.fillStyle = 'rgba(165, 243, 252, 0.95)';
            c.font = '12px ui-monospace, monospace';
            const hd = handles().find((q) => q.rot);
            if (hd) c.fillText(`${Math.round((drag.angle * 180) / Math.PI)}°`, hd.x + 10, hd.y - 8);
          }
        }
        if (paint) {
          c.fillStyle = paint.sub ? 'rgba(248, 113, 113, 0.18)' : 'rgba(250, 204, 21, 0.16)';
          const rs = paint.r * E.cam.zoom;
          c.beginPath();
          for (const [x, y] of paint.pts) { const [sx, sy] = E.toScreen(x, y); c.moveTo(sx + rs, sy); c.arc(sx, sy, rs, 0, Math.PI * 2); }
          c.fill();
        }
        c.setLineDash([]);
        if (band) {
          c.strokeStyle = 'rgba(165, 243, 252, 0.8)';
          c.fillStyle = 'rgba(0, 242, 254, 0.06)';
          const x = Math.min(band.x0, band.x1), y = Math.min(band.y0, band.y1), w = Math.abs(band.x1 - band.x0), h = Math.abs(band.y1 - band.y0);
          c.fillRect(x, y, w, h);
          c.strokeRect(x, y, w, h);
        }
        if (tool === 'points') {
          for (const o of selected())
            o.strokes.forEach((s, si) => s.pts.forEach((p, pi) => {
              const [sx, sy] = E.toScreen(p[0], p[1]);
              if (sx < -8 || sy < -8 || sx > W + 8 || sy > H + 8) return;
              const on = point && point.o === o && point.si === si && point.pi === pi;
              const hov = hoverPt && hoverPt.o === o && hoverPt.si === si && hoverPt.pi === pi;
              c.fillStyle = on ? 'rgba(250, 204, 21, 0.95)' : hov ? 'rgba(240, 255, 255, 0.95)' : 'rgba(0, 242, 254, 0.75)';
              c.fillRect(sx - (on || hov ? 4 : 2.5), sy - (on || hov ? 4 : 2.5), on || hov ? 8 : 5, on || hov ? 8 : 5);
            }));
        }
      }
      if (tool === 'area' && brushAt) {
        c.strokeStyle = 'rgba(250, 204, 21, 0.75)';
        c.beginPath();
        c.arc(brushAt[0], brushAt[1], Math.max(1, +brushR.value), 0, Math.PI * 2);
        c.stroke();
      }
      if (tool === 'brush' && brushAt) {
        // кружок кисти: видимая толщина мазка (радиус × толщина объекта)
        const o = first(), th = o ? o.get('thick') : M.STYLE.thick[2];
        c.strokeStyle = 'rgba(165, 243, 252, 0.7)';
        c.beginPath();
        c.arc(brushAt[0], brushAt[1], Math.max(1, +brushR.value * th), 0, Math.PI * 2);
        c.stroke();
      }
      drawMini(c);
    });

    /** Мини-карта: весь мир (рамка сцены и камера), объекты, видимая часть — рамкой. Клик переносит камеру. */
    function drawMini(c) {
      const mr = miniRect();
      const view = E.cam.view(E.W, E.H);
      const world = M.union(E.scene.bounds(), view);
      if (M.isEmpty(world)) { miniMap = null; return; }
      const ww = world.x1 - world.x0, wh = world.y1 - world.y0, s = Math.min((mr.w - 8) / ww, (mr.h - 8) / wh);
      miniMap = { s, ox: (mr.w - ww * s) / 2 - world.x0 * s, oy: (mr.h - wh * s) / 2 - world.y0 * s };
      const X = (x) => mr.x + miniMap.ox + x * s, Y = (y) => mr.y + miniMap.oy + y * s;
      c.fillStyle = 'rgba(2, 8, 20, 0.82)';
      c.strokeStyle = 'rgba(0, 242, 254, 0.28)';
      c.fillRect(mr.x, mr.y, mr.w, mr.h);
      c.strokeRect(mr.x + 0.5, mr.y + 0.5, mr.w - 1, mr.h - 1);
      for (const o of E.objects) {
        const b = o.bounds();
        if (M.isEmpty(b)) continue;
        c.fillStyle = selIds.includes(o.id) ? 'rgba(250, 204, 21, 0.8)' : 'rgba(0, 242, 254, 0.45)';
        c.fillRect(X(b.x0), Y(b.y0), Math.max(1.5, (b.x1 - b.x0) * s), Math.max(1.5, (b.y1 - b.y0) * s));
      }
      c.strokeStyle = 'rgba(240, 255, 255, 0.9)';
      c.strokeRect(X(view.x0), Y(view.y0), (view.x1 - view.x0) * s, (view.y1 - view.y0) * s);
    }

    fillPick();
    showFields();
    updateHist();
    setTool('view');
  }

  function inPoly(poly, x, y) {
    let c = false;
    for (let i = 0, j = poly.length - 1; i < poly.length; j = i++) {
      const [xi, yi] = poly[i], [xj, yj] = poly[j];
      if (yi > y !== yj > y && x < ((xj - xi) * (y - yi)) / (yj - yi || 1e-9) + xi) c = !c;
    }
    return c;
  }

  function smoothPts(pts, win) {
    if (win < 1 || pts.length < 3) return pts;
    const out = [pts[0]], D = pts[0].length;
    for (let i = 1; i < pts.length - 1; i++) {
      const acc = new Array(D).fill(0);
      let c = 0;
      for (let j = -win; j <= win; j++) {
        const p = pts[Math.min(pts.length - 1, Math.max(0, i + j))];
        for (let d = 0; d < D; d++) acc[d] += p[d];
        c++;
      }
      out.push(acc.map((v) => v / c));
    }
    out.push(pts[pts.length - 1]);
    return out;
  }

  /*
   * Область из мазков: прежняя область объекта и кружки мазка (радиус кисти) — на сетку мира, Alt-мазок стирает.
   * Контур закрашенного — квадраты-марши по сетке; берётся самый большой контур, упрощается до 200 точек.
   */
  function paintArea(prev, pts, r, sub) {
    let x0 = Infinity, y0 = Infinity, x1 = -Infinity, y1 = -Infinity;
    const grow = (x, y, m) => { x0 = Math.min(x0, x - m); y0 = Math.min(y0, y - m); x1 = Math.max(x1, x + m); y1 = Math.max(y1, y + m); };
    if (prev) for (const [x, y] of prev) grow(x, y, 0);
    if (!sub) for (const [x, y] of pts) grow(x, y, r);
    if (!(x1 > x0)) return null;
    const cell = Math.max(r / 3, Math.max(x1 - x0, y1 - y0) / 300);
    const ox = x0 - cell * 2, oy = y0 - cell * 2;
    const cols = Math.ceil((x1 - x0) / cell) + 5, rows = Math.ceil((y1 - y0) / cell) + 5;
    const g = new Uint8Array(cols * rows);
    if (prev && prev.length >= 3)
      for (let j = 1; j < rows - 1; j++) for (let i = 1; i < cols - 1; i++) if (inPoly(prev, ox + i * cell, oy + j * cell)) g[j * cols + i] = 1;
    const rc = Math.ceil(r / cell);
    for (const [x, y] of pts) {
      const ci = Math.round((x - ox) / cell), cj = Math.round((y - oy) / cell);
      for (let j = Math.max(1, cj - rc); j <= Math.min(rows - 2, cj + rc); j++)
        for (let i = Math.max(1, ci - rc); i <= Math.min(cols - 2, ci + rc); i++)
          if (Math.hypot(ox + i * cell - x, oy + j * cell - y) <= r) g[j * cols + i] = sub ? 0 : 1;
    }
    const loops = marchLoops(g, cols, rows);
    let best = null, bestA = 0;
    for (const L of loops) {
      let a = 0;
      for (let k = 0, m = L.length - 1; k < L.length; m = k++) a += L[m][0] * L[k][1] - L[k][0] * L[m][1];
      if (Math.abs(a) > bestA) { bestA = Math.abs(a); best = L; }
    }
    if (!best || best.length < 3) return null;
    // контур по сетке ступенчатый: два прохода сглаживания по кругу, потом упрощение
    let out = best.map(([i, j]) => [ox + i * cell, oy + j * cell]), eps = cell * 0.35;
    for (let pass = 0; pass < 2; pass++) out = out.map((q, k) => { const a = out[(k + out.length - 1) % out.length], c = out[(k + 1) % out.length]; return [(a[0] + 2 * q[0] + c[0]) / 4, (a[1] + 2 * q[1] + c[1]) / 4]; });
    let simple = rdp(out, eps);
    while (simple.length > 200) { eps *= 1.4; simple = rdp(out, eps); }
    return simple.length >= 3 ? simple.map((q) => q.map((v) => Math.round(v * 10) / 10)) : null;
  }
  /** Контуры закрашенных клеток: середины рёбер сетки, у каждой ровно два соседа — обход по кругу. */
  function marchLoops(g, cols, rows) {
    const H = (i, j) => (j * cols + i) * 2, V = (i, j) => (j * cols + i) * 2 + 1;
    const adj = new Map();
    const link = (a, b) => {
      (adj.get(a) || adj.set(a, []).get(a)).push(b);
      (adj.get(b) || adj.set(b, []).get(b)).push(a);
    };
    for (let j = 0; j < rows - 1; j++)
      for (let i = 0; i < cols - 1; i++) {
        const c = (g[j * cols + i] << 3) | (g[j * cols + i + 1] << 2) | (g[(j + 1) * cols + i + 1] << 1) | g[(j + 1) * cols + i];
        if (!c || c === 15) continue;
        const T = H(i, j), B = H(i, j + 1), L = V(i, j), R = V(i + 1, j);
        const segs = { 1: [[L, B]], 2: [[B, R]], 3: [[L, R]], 4: [[T, R]], 5: [[T, R], [L, B]], 6: [[T, B]], 7: [[T, L]], 8: [[T, L]],
          9: [[T, B]], 10: [[T, L], [B, R]], 11: [[T, R]], 12: [[L, R]], 13: [[B, R]], 14: [[L, B]] }[c];
        for (const [a, b] of segs) link(a, b);
      }
    const at = (id) => { const e = id >> 1, i = e % cols, j = (e / cols) | 0; return id & 1 ? [i, j + 0.5] : [i + 0.5, j]; };
    const seen = new Set(), loops = [];
    for (const startId of adj.keys()) {
      if (seen.has(startId)) continue;
      const loop = [];
      let prev = -1, cur = startId;
      while (cur !== undefined && !seen.has(cur)) {
        seen.add(cur);
        loop.push(at(cur));
        const nb = adj.get(cur);
        const next = nb[0] !== prev ? nb[0] : nb[1];
        prev = cur;
        cur = next;
      }
      if (loop.length >= 3) loops.push(loop);
    }
    return loops;
  }

  function rdp(pts, eps) {
    if (pts.length < 3) return pts;
    const keep = new Uint8Array(pts.length);
    keep[0] = keep[pts.length - 1] = 1;
    const stack = [[0, pts.length - 1]];
    while (stack.length) {
      const [a, b] = stack.pop();
      const ax = pts[a][0], ay = pts[a][1], ex = pts[b][0] - ax, ey = pts[b][1] - ay, el = Math.hypot(ex, ey) || 1e-9;
      let md = -1, mi = -1;
      for (let i = a + 1; i < b; i++) {
        const d = Math.abs((pts[i][0] - ax) * ey - (pts[i][1] - ay) * ex) / el;
        if (d > md) { md = d; mi = i; }
      }
      if (md > eps) { keep[mi] = 1; stack.push([a, mi], [mi, b]); }
    }
    return pts.filter((_, i) => keep[i]);
  }
