// Редактор витрины хранителя. Отдаётся как /k/editor.js только при действующей сессии (__Host-session):
// без сессии этого кода в браузере нет. Работает поверх движка /vitrina.js (событие vitrina:ready), модель — E.M.
// Холст как в графических редакторах: пробел + перетаскивание или средняя кнопка — сдвиг, колёсико — сдвиг,
// Ctrl/⌘ + колёсико — масштаб, Shift+1 — вся сцена, Shift+2 — к выделенному, Ctrl+Z / Ctrl+Shift+Z — отмена и повтор.
// Инструменты: смотреть (как гость), выбор (клик по объекту или внутри его рамки, Shift — добавить, рамка, перетаскивание;
//   у рамки выделенного — ручки: углы и стороны растягивают (Shift — пропорционально, Alt — от центра), кружок сверху поворачивает,
//   Shift — шагами по 15°),
// точки (перетащить точку жилы, клик по жиле — новая точка, двойной клик или Delete — убрать точку),
// кисть (мазок в выбранный объект), ластик (клик по жиле), область (клики — вершины, двойной клик или Enter — замкнуть).
// Панель свойств выделенного объекта: подпись, адрес, слой, параметры жил (tubeR, толщина, резонанс, такты…), окно.
// Кнопки «Сохранить черновик», «Опубликовать», «Откатить» — POST в /k/vitrina/* с CSRF-токеном сессии.
(() => {
  'use strict';
  const csrf = (document.querySelector('meta[name="k-csrf"]') || {}).content || '';
  const start = (api) => {
    if (start.done) return;
    start.done = true;
    init(api);
  };
  if (window.vitrinaEngine) start(window.vitrinaEngine);
  document.addEventListener('vitrina:ready', (e) => start(e.detail));

  function init(E) {
    const M = E.M, cv = E.cv;
    const version = cv.dataset.version || '';
    const el = (tag, cls, text) => {
      const n = document.createElement(tag);
      if (cls) n.className = cls;
      if (text) n.textContent = text;
      return n;
    };
    const button = (parent, name, fn, cls, title) => {
      const b = el('button', 'kp-b' + (cls ? ' ' + cls : ''), name);
      b.type = 'button';
      if (title) b.title = title;
      b.addEventListener('click', fn);
      parent.append(b);
      return b;
    };
    const input = (tag) => el(tag || 'input', 'kp-in');
    const wrap = (inp, text) => {
      const l = el('label', 'kp-f');
      l.append(el('span', '', text), inp);
      return l;
    };
    const section = (title, open) => {
      const d = el('details', 'kp-sec');
      d.open = !!open;
      d.append(el('summary', '', title));
      return d;
    };

    const panel = el('div', 'kp');
    const head = el('p', 'kp-note', version ? `Предпросмотр версии ${version}` : 'Витрина: опубликованная сцена');
    const toolsRow = el('div', 'kp-row');
    const TOOLS = { view: 'Смотреть', select: 'Выбор', points: 'Точки', brush: 'Кисть', erase: 'Ластик', area: 'Область' };
    const btns = {};
    for (const [t, name] of Object.entries(TOOLS)) btns[t] = button(toolsRow, name, () => setTool(t));
    const histRow = el('div', 'kp-row');
    const bUndo = button(histRow, 'Отменить', undo, '', 'Ctrl+Z');
    const bRedo = button(histRow, 'Повторить', redo, '', 'Ctrl+Shift+Z');
    button(histRow, 'Вся сцена', () => E.fitAll(), '', 'Shift+1');
    button(histRow, 'К выделенному', fitSelection, '', 'Shift+2');
    const pick = el('select', 'kp-sel');
    const objRow = el('div', 'kp-row');

    const props = section('Объект', true);
    const fId = input(), fLabel = input(), fHref = input(), fZ = input(), fTitle = input(), fText = input('textarea'), fLinks = input('textarea');

      wrap(fId, 'id: латиница, цифры, дефис'),

      wrap(fHref, 'Адрес: относительный путь, например /wall'),

    function slider(parent, k, name, lo, hi, kind) {
      const row = el('label', 'kp-f kp-sl');
      const top = el('span', '', name);
      const out = el('output', 'kp-out');
      const reset = el('button', 'kp-x', '×');
      reset.type = 'button';
      reset.title = 'по умолчанию';
      top.append(out, reset);

        const t = target();

      hist.commit(snap());
      updateHist();

      bUndo.disabled = !hist.canUndo;
      bRedo.disabled = !hist.canRedo;
    }

      if (!state) return;
      E.load(JSON.parse(state));
      selIds = selIds.filter(byId);
      fillPick();
      showFields();
      updateHist();
    }
    function undo() { restore(hist.undo()); }
    function redo() { restore(hist.redo()); }

      tool = t;
      E.setTool(t);
      for (const [k, b] of Object.entries(btns)) b.classList.toggle('on', k === t);
      if (t !== 'area') areaPts = [];
      point = null;
    }

      if (!M.isEmpty(b)) E.flyTo(M.expand(b, Math.max(b.x1 - b.x0, b.y1 - b.y0) * 0.15));
    }

      pick.textContent = '';
      pick.append(new Option('— ничего не выбрано —', ''));
      E.objects.forEach((o) => pick.append(new Option(`${o.id} — ${o.label}`, o.id)));
      pick.value = selIds[0] || '';

    function showFields() {
      const o = first();
      for (const f of [fId, fLabel, fHref, fZ, fTitle, fText, fLinks]) f.disabled = !o;
      for (const s of Object.values(sliders)) s.r.disabled = !o;
      fId.value = o ? o.id : '';
      fLabel.value = o ? o.label : '';
      fHref.value = (o && o.href) || '';
      fZ.value = o ? String(o.z || 0) : '';
      fTitle.value = (o && o.window && o.window.title) || '';
      fText.value = (o && o.window && o.window.text) || '';
      fLinks.value = ((o && o.window && o.window.links) || []).map((l) => `${l.label} | ${l.href}`).join('\n');
      for (const s of Object.values(sliders)) {
        const own = !!(o && o.style && typeof o.style[s.k] === 'number');
        const v = o ? (s.k === 'tubeR' ? o.tubeR() : o.get(s.k)) : M.STYLE[s.k][2];
        s.r.value = String(v);
        s.out.textContent = String(Math.round(v * 1000) / 1000) + (own ? '' : ' (общий)');
        s.reset.hidden = !own;
      }
      for (const s of Object.values(spaceSliders)) {
        const own = !!(E.scene.space && typeof E.scene.space[s.k] === 'number');
        const v = E.scene.get(s.k);
        s.r.value = String(v);
        s.out.textContent = String(v) + (own ? '' : ' (общий)');
        s.reset.hidden = !own;
      }
      props.querySelector('summary').textContent = selIds.length > 1 ? `Объект (выбрано ${selIds.length}, свойства первого)` : 'Объект';

    function readFields() {

      if (!o) return;

      o.href = fHref.value.trim() || null;
      o.z = Math.max(-100, Math.min(100, Math.round(+fZ.value || 0)));

        .split('\n')
        .map((l) => l.split('|').map((x) => x.trim()))
        .filter((l) => l[0] && l[1])
        .map(([label, href]) => ({ label, href }));
      const w = {};
      if (fTitle.value.trim()) w.title = fTitle.value.trim();
      if (fText.value.trim()) w.text = fText.value.trim();

      o.window = Object.keys(w).length ? w : null;
      const opt = [...pick.options].find((x) => x.value === o.id);
      if (opt) opt.textContent = `${o.id} — ${o.label}`;

      changed(true);

    let drag = null, raw = [], areaPts = [], band = null, point = null, hoverPt = null;

    /** Ближайшая точка жилы выделенных объектов в пределах d пикселей экрана. */
    function nearPoint(x, y, d) {
      let best = null;
      for (const o of selected())
        o.strokes.forEach((s, si) => s.pts.forEach((p, pi) => {
          const [sx, sy] = E.toScreen(p[0], p[1]), dd = Math.hypot(sx - x, sy - y);
          if (dd < d && (!best || dd < best.d)) best = { o, si, pi, d: dd };
        }));
      return best;
    }
    /** Ближайший отрезок жилы выделенных объектов: куда вставить новую точку. */
    function nearSegment(x, y, d) {
      let best = null;
      for (const o of selected())
        o.strokes.forEach((s, si) => {
          const pts = s.closed ? [...s.pts, s.pts[0]] : s.pts;
          for (let i = 0; i < pts.length - 1; i++) {
            const [ax, ay] = E.toScreen(pts[i][0], pts[i][1]), [bx, by] = E.toScreen(pts[i + 1][0], pts[i + 1][1]);
            const ex = bx - ax, ey = by - ay, t = Math.max(0, Math.min(1, ((x - ax) * ex + (y - ay) * ey) / (ex * ex + ey * ey || 1e-9)));
            const dd = Math.hypot(x - ax - ex * t, y - ay - ey * t);
            if (dd < d && (!best || dd < best.d)) best = { o, si, at: i + 1, d: dd };
          }
        });
      return best;
    }
    /** Объект под точкой для выбора: по жилам и областям, иначе — самый маленький, чья рамка содержит точку. */
    function pickAt(x, y) {
      const i = E.hit(x, y);
      if (i >= 0) return i;
      const [wx, wy] = E.toWorld(x, y);
      let best = -1, area = Infinity;
      E.objects.forEach((o, j) => {
        const b = o.bounds(), a = (b.x1 - b.x0) * (b.y1 - b.y0);
        if (wx >= b.x0 && wx <= b.x1 && wy >= b.y0 && wy <= b.y1 && a < area) { best = j; area = a; }
      });
      return best;
    }
    /* ---------- ручки рамки выделенного: растяжение и поворот ---------- */
    const HANDLE = 6;
    function selBox() {
      const b = selected().reduce((a, o) => M.union(a, o.bounds()), { ...M.EMPTY });
      if (M.isEmpty(b)) return null;
      const [x0, y0] = E.toScreen(b.x0, b.y0), [x1, y1] = E.toScreen(b.x1, b.y1);
      return { b, x0: x0 - 3, y0: y0 - 3, x1: x1 + 3, y1: y1 + 3 };
    }
    /** Ручки в пикселях экрана: fx, fy — доля рамки (0, ½, 1); rot — поворот. */
    function handles() {
      const s = selBox();
      if (!s) return [];
      const out = [];
      for (const fy of [0, 0.5, 1]) for (const fx of [0, 0.5, 1]) {
        if (fx === 0.5 && fy === 0.5) continue;
        out.push({ fx, fy, x: s.x0 + (s.x1 - s.x0) * fx, y: s.y0 + (s.y1 - s.y0) * fy });
      }
      out.push({ rot: true, x: (s.x0 + s.x1) / 2, y: s.y0 - 22 });
      return out;
    }
    function handleAt(x, y) {
      return handles().find((h) => Math.abs(h.x - x) <= HANDLE + 2 && Math.abs(h.y - y) <= HANDLE + 2) || null;
    }
    /** Начало правки ручкой: снимок жил и областей, чтобы каждый шаг считался от исходного вида без накопления округлений. */
    function startHandle(h, wx, wy) {
      const b = selected().reduce((a, o) => M.union(a, o.bounds()), { ...M.EMPTY });
      const orig = selected().map((o) => ({ o, strokes: JSON.stringify(o.strokes), area: o.area ? JSON.stringify(o.area) : null }));
      drag = { kind: h.rot ? 'rotate' : 'stretch', h, b, orig, start: [wx, wy], moved: false };
    }
    function applyHandle(e, wx, wy) {
      const { h, b, orig } = drag;
      for (const r of orig) { r.o.strokes = JSON.parse(r.strokes); r.o.area = r.area ? JSON.parse(r.area) : null; }
      const cx = (b.x0 + b.x1) / 2, cy = (b.y0 + b.y1) / 2;
      if (drag.kind === 'rotate') {
        let a = Math.atan2(wy - cy, wx - cx) - Math.atan2(drag.start[1] - cy, drag.start[0] - cx);
        if (e.shiftKey) a = Math.round(a / (Math.PI / 12)) * (Math.PI / 12);
        for (const r of orig) r.o.rotate(a, cx, cy);
        drag.angle = a;
      } else {
        // опора — противоположная сторона (Alt — центр рамки)
        const hx = b.x0 + (b.x1 - b.x0) * h.fx, hy = b.y0 + (b.y1 - b.y0) * h.fy;
        const px = e.altKey ? cx : b.x0 + (b.x1 - b.x0) * (1 - h.fx), py = e.altKey ? cy : b.y0 + (b.y1 - b.y0) * (1 - h.fy);
        const lim = (v) => (Math.abs(v) < 0.02 ? (v < 0 ? -0.02 : 0.02) : v);
        let sx = h.fx === 0.5 ? 1 : lim((wx - px) / (hx - px || 1e-9));
        let sy = h.fy === 0.5 ? 1 : lim((wy - py) / (hy - py || 1e-9));
        if (e.shiftKey) {
          const k = h.fx === 0.5 ? sy : h.fy === 0.5 ? sx : Math.sign(sx) * Math.max(Math.abs(sx), Math.abs(sy));
          sx = Math.sign(sx || 1) * Math.abs(k); sy = Math.sign(sy || 1) * Math.abs(k);
        }
        for (const r of orig) r.o.stretch(sx, sy, px, py);
        drag.k = [sx, sy];
      }
      drag.moved = true;
      E.rebuild(2);
    }

    function deletePoint(pt) {
      const s = pt.o.strokes[pt.si];
      s.pts.splice(pt.pi, 1);
      if (!s.pts.length) pt.o.strokes.splice(pt.si, 1);
      point = null;
      changed(true);
    }

    E.setInput({
      down(e) {
        const x = e.clientX, y = e.clientY, [wx, wy] = E.toWorld(x, y);
        const mr = miniRect();
        if (miniMap && x >= mr.x && x <= mr.x + mr.w && y >= mr.y && y <= mr.y + mr.h) {
          drag = { kind: 'mini' };
          miniJump(x, y);
          return;
        }
        if (tool === 'select') {
          const h = selIds.length ? handleAt(x, y) : null;
          if (h) { startHandle(h, wx, wy); return; }
          const i = pickAt(x, y);
          if (i >= 0) {
            const id = E.objects[i].id;
            if (e.shiftKey) select(selIds.includes(id) ? selIds.filter((s) => s !== id) : [...selIds, id]);
            else if (!selIds.includes(id)) select([id]);
            drag = { kind: 'move', last: [wx, wy], moved: false };
          } else {
            if (!e.shiftKey) select([]);
            band = { x0: x, y0: y, x1: x, y1: y, add: e.shiftKey };
          }
        } else if (tool === 'points') {
          const p = nearPoint(x, y, 8);
          if (p) { point = p; drag = { kind: 'point', moved: false }; return; }
          const sgm = nearSegment(x, y, 8);
          if (sgm) {
            const s = sgm.o.strokes[sgm.si];
            s.pts.splice(sgm.at, 0, [r1(wx), r1(wy)]);
            point = { o: sgm.o, si: sgm.si, pi: sgm.at };
            drag = { kind: 'point', moved: true };
            E.rebuild(2);
            return;
          }
          const i = pickAt(x, y);
          select(i >= 0 ? [E.objects[i].id] : []);
          point = null;
        } else if (tool === 'brush') {
          let o = first();
          if (!o) {
            const id = E.scene.newId();
            o = new M.SceneObject({ id, label: 'Новый объект', strokes: [] });
            E.objects.push(o);
            selIds = [id];
            fillPick();
            showFields();
          }
          raw = [[wx, wy]];
          E.setLive({ obj: E.objects.indexOf(o), st: { r: o.tubeR(), pts: raw.slice() } });
        } else if (tool === 'erase') {
          const n = E.nearest(x, y);
          if (!n || n.sd > 14 || n.si < 0) return;
          E.objects[n.obj].strokes.splice(n.si, 1);
          changed(true);
        } else if (tool === 'area') {
          if (!first()) { const i = pickAt(x, y); if (i >= 0) select([E.objects[i].id]); else { note.textContent = 'Сначала выберите объект для области.'; return; } }
          areaPts.push([r1(wx), r1(wy)]);
        }
      },
      move(e) {
        const x = e.clientX, y = e.clientY, [wx, wy] = E.toWorld(x, y);
        if (tool === 'points' && !drag) hoverPt = nearPoint(x, y, 8);
        if (!drag && !band && !raw.length) return;
        if (drag && drag.kind === 'mini') miniJump(x, y);
        else if (drag && (drag.kind === 'stretch' || drag.kind === 'rotate')) applyHandle(e, wx, wy);
        else if (drag && drag.kind === 'move') {
          const dx = wx - drag.last[0], dy = wy - drag.last[1];
          if (!dx && !dy) return;
          for (const o of selected()) o.translate(dx, dy);
          drag.last = [wx, wy];
          drag.moved = true;
          E.rebuild(2);
        } else if (drag && drag.kind === 'point' && point) {
          const p = point.o.strokes[point.si].pts[point.pi];
          p[0] = r1(wx); p[1] = r1(wy);
          drag.moved = true;
          E.rebuild(2);
        } else if (band) {
          band.x1 = x; band.y1 = y;
        } else if (raw.length) {
          const l = raw[raw.length - 1];
          if (Math.hypot(wx - l[0], wy - l[1]) * E.cam.zoom > 1.5) {
            raw.push([wx, wy]);
            const o = first();
            E.setLive({ obj: E.objects.indexOf(o), st: { r: o.tubeR(), pts: smoothPts(raw, 2) } });
          }
        }
      },
      up() {
        if (drag && /^(move|point|stretch|rotate)$/.test(drag.kind) && drag.moved) changed(true);
        drag = null;
        if (band) {
          const [ax, ay] = E.toWorld(band.x0, band.y0), [bx, by] = E.toWorld(band.x1, band.y1);
          if (Math.hypot(band.x1 - band.x0, band.y1 - band.y0) > 4) {
            const rect = M.rectFrom(ax, ay, bx, by);
            const ids = E.scene.visible(rect).map((i) => E.objects[i].id);
            select(band.add ? [...selIds, ...ids] : ids);
          }
          band = null;
        }
        if (raw.length) {
          const o = first();
          if (raw.length > 1 && o) {
            const eps = 0.35 / E.cam.zoom;
            const pts = rdp(smoothPts(smoothPts(raw, 4), 2), eps).map((p) => p.map(r1));
            o.strokes.push({ r: o.tubeR(), pts });
          }
          raw = [];
          E.setLive(null);
          if (o && !o.strokes.length && !o.area) E.objects.splice(E.objects.indexOf(o), 1);
          changed(true);
        }
      },
      cancel() {
        if (drag && drag.orig) { for (const r of drag.orig) { r.o.strokes = JSON.parse(r.strokes); r.o.area = r.area ? JSON.parse(r.area) : null; } E.rebuild(2); }
        drag = null; band = null;
        if (raw.length) { raw = []; E.setLive(null); }
      },
      dblclick(e) {
        if (tool === 'area') closeArea();
        else if (tool === 'points') {
          const p = nearPoint(e.clientX, e.clientY, 8);
          if (p) deletePoint(p);
        } else if (tool === 'select' && pickAt(e.clientX, e.clientY) >= 0) fitSelection();
      },
    });

    function closeArea() {
      const o = first();
      if (!o || areaPts.length < 3) return;
      o.area = areaPts.slice(0, 200);
      areaPts = [];
      changed(true);
    }
    function miniJump(x, y) {
      const mr = miniRect();
      E.cam.x = (x - mr.x - miniMap.ox) / miniMap.s;
      E.cam.y = (y - mr.y - miniMap.oy) / miniMap.s;
      E.cameraChanged();
    }

    window.addEventListener('keydown', (e) => {
      if (e.target && /^(INPUT|TEXTAREA|SELECT)$/.test(e.target.tagName)) return;
      const mod = e.ctrlKey || e.metaKey;
      if (mod && (e.key === 'z' || e.key === 'Z' || e.code === 'KeyZ')) { e.preventDefault(); e.shiftKey ? redo() : undo(); return; }
      if (mod && (e.key === 'y' || e.code === 'KeyY')) { e.preventDefault(); redo(); return; }
      if (e.code === 'Digit2' && e.shiftKey) { e.preventDefault(); fitSelection(); return; }
      if (tool === 'view') return;
      if (e.key === 'Enter' && tool === 'area') closeArea();
      if (e.key === 'Escape') { areaPts = []; band = null; point = null; if (tool === 'select') select([]); }
      if (e.key === 'Delete' || e.key === 'Backspace') {
        e.preventDefault();
        if (tool === 'points' && point) deletePoint(point);
        else if (tool === 'select') deleteSelected();
      }
    });

    /* ---------- поверх кадра: выделение, точки, область, рамка выбора, мини-карта ---------- */
    E.setOverlay((c) => {
      const W = E.W, H = E.H;
      c.lineWidth = 1;
      if (tool !== 'view') {
        c.setLineDash([3, 6]);
        for (const o of selected()) {
          const b = o.bounds();
          if (M.isEmpty(b)) continue;
          const [x0, y0] = E.toScreen(b.x0, b.y0), [x1, y1] = E.toScreen(b.x1, b.y1);
          c.strokeStyle = 'rgba(0, 242, 254, 0.6)';
          c.strokeRect(x0 - 3, y0 - 3, x1 - x0 + 6, y1 - y0 + 6);
          if (o.area) {
            c.strokeStyle = 'rgba(250, 204, 21, 0.7)';
            c.beginPath();
            o.area.forEach(([x, y], j) => { const [sx, sy] = E.toScreen(x, y); j ? c.lineTo(sx, sy) : c.moveTo(sx, sy); });
            c.closePath();
            c.stroke();
          }
        }
        if (tool === 'select' && selIds.length) {
          for (const hd of handles()) {
            c.fillStyle = 'rgba(2, 8, 20, 0.9)';
            c.strokeStyle = 'rgba(165, 243, 252, 0.95)';
            c.beginPath();
            if (hd.rot) c.arc(hd.x, hd.y, HANDLE - 1, 0, Math.PI * 2);
            else c.rect(hd.x - HANDLE / 2 - 1, hd.y - HANDLE / 2 - 1, HANDLE + 2, HANDLE + 2);
            c.fill();
            c.stroke();
          }
          if (drag && drag.kind === 'rotate' && typeof drag.angle === 'number') {
            c.fillStyle = 'rgba(165, 243, 252, 0.95)';
            c.font = '12px ui-monospace, monospace';
            const hd = handles().find((q) => q.rot);
            if (hd) c.fillText(`${Math.round((drag.angle * 180) / Math.PI)}°`, hd.x + 10, hd.y - 8);
          }
        }
        if (areaPts.length) {
          c.strokeStyle = 'rgba(250, 204, 21, 0.9)';
          c.beginPath();
          areaPts.forEach(([x, y], j) => { const [sx, sy] = E.toScreen(x, y); j ? c.lineTo(sx, sy) : c.moveTo(sx, sy); });
          c.stroke();
        }
        c.setLineDash([]);
        if (band) {
          c.strokeStyle = 'rgba(165, 243, 252, 0.8)';
          c.fillStyle = 'rgba(0, 242, 254, 0.06)';
          const x = Math.min(band.x0, band.x1), y = Math.min(band.y0, band.y1), w = Math.abs(band.x1 - band.x0), h = Math.abs(band.y1 - band.y0);
          c.fillRect(x, y, w, h);
          c.strokeRect(x, y, w, h);
        }
        if (tool === 'points') {
          for (const o of selected())
            o.strokes.forEach((s, si) => s.pts.forEach((p, pi) => {
              const [sx, sy] = E.toScreen(p[0], p[1]);
              if (sx < -8 || sy < -8 || sx > W + 8 || sy > H + 8) return;
              const on = point && point.o === o && point.si === si && point.pi === pi;
              const hov = hoverPt && hoverPt.o === o && hoverPt.si === si && hoverPt.pi === pi;
              c.fillStyle = on ? 'rgba(250, 204, 21, 0.95)' : hov ? 'rgba(240, 255, 255, 0.95)' : 'rgba(0, 242, 254, 0.75)';
              c.fillRect(sx - (on || hov ? 4 : 2.5), sy - (on || hov ? 4 : 2.5), on || hov ? 8 : 5, on || hov ? 8 : 5);
            }));
        }
      }
      drawMini(c);
    });

    /** Мини-карта: весь мир (рамка сцены и камера), объекты, видимая часть — рамкой. Клик переносит камеру. */
    function drawMini(c) {
      const mr = miniRect();
      const view = E.cam.view(E.W, E.H);
      const world = M.union(E.scene.bounds(), view);
      if (M.isEmpty(world)) { miniMap = null; return; }
      const ww = world.x1 - world.x0, wh = world.y1 - world.y0, s = Math.min((mr.w - 8) / ww, (mr.h - 8) / wh);
      miniMap = { s, ox: (mr.w - ww * s) / 2 - world.x0 * s, oy: (mr.h - wh * s) / 2 - world.y0 * s };
      const X = (x) => mr.x + miniMap.ox + x * s, Y = (y) => mr.y + miniMap.oy + y * s;
      c.fillStyle = 'rgba(2, 8, 20, 0.82)';
      c.strokeStyle = 'rgba(0, 242, 254, 0.28)';
      c.fillRect(mr.x, mr.y, mr.w, mr.h);
      c.strokeRect(mr.x + 0.5, mr.y + 0.5, mr.w - 1, mr.h - 1);
      for (const o of E.objects) {
        const b = o.bounds();
        if (M.isEmpty(b)) continue;
        c.fillStyle = selIds.includes(o.id) ? 'rgba(250, 204, 21, 0.8)' : 'rgba(0, 242, 254, 0.45)';
        c.fillRect(X(b.x0), Y(b.y0), Math.max(1.5, (b.x1 - b.x0) * s), Math.max(1.5, (b.y1 - b.y0) * s));
      }
      c.strokeStyle = 'rgba(240, 255, 255, 0.9)';
      c.strokeRect(X(view.x0), Y(view.y0), (view.x1 - view.x0) * s, (view.y1 - view.y0) * s);
    }

    fillPick();
    showFields();
    updateHist();
    setTool('view');
  }

  function smoothPts(pts, win) {
    if (win < 1 || pts.length < 3) return pts;
    const out = [pts[0]], D = pts[0].length;
    for (let i = 1; i < pts.length - 1; i++) {
      const acc = new Array(D).fill(0);
      let c = 0;
      for (let j = -win; j <= win; j++) {
        const p = pts[Math.min(pts.length - 1, Math.max(0, i + j))];
        for (let d = 0; d < D; d++) acc[d] += p[d];
        c++;
      }
      out.push(acc.map((v) => v / c));
    }
    out.push(pts[pts.length - 1]);
    return out;
  }

  function rdp(pts, eps) {
    if (pts.length < 3) return pts;
    const keep = new Uint8Array(pts.length);
    keep[0] = keep[pts.length - 1] = 1;
    const stack = [[0, pts.length - 1]];
    while (stack.length) {
      const [a, b] = stack.pop();
      const ax = pts[a][0], ay = pts[a][1], ex = pts[b][0] - ax, ey = pts[b][1] - ay, el = Math.hypot(ex, ey) || 1e-9;
      let md = -1, mi = -1;
      for (let i = a + 1; i < b; i++) {
        const d = Math.abs((pts[i][0] - ax) * ey - (pts[i][1] - ay) * ex) / el;
        if (d > md) { md = d; mi = i; }
      }
      if (md > eps) { keep[mi] = 1; stack.push([a, mi], [mi, b]); }
    }
    return pts.filter((_, i) => keep[i]);
  }
