// Геометрия редактора: сглаживание и упрощение мазков, область из мазков кисти (сетка + квадраты-марши).
// Без DOM и движка: функции чистые.

export { inPoly } from '/vitrina/model.js';
import { inPoly } from '/vitrina/model.js';

export const r1 = (v) => Math.round(v * 10) / 10;
export const r2 = (v) => Math.round(v * 100) / 100;

/** Скользящее среднее по точкам, концы на месте. */
export function smoothPts(pts, win) {
  if (win < 1 || pts.length < 3) return pts;
  const out = [pts[0]], D = pts[0].length;
  for (let i = 1; i < pts.length - 1; i++) {
    const acc = new Array(D).fill(0);
    let c = 0;
    for (let j = -win; j <= win; j++) {
      const p = pts[Math.min(pts.length - 1, Math.max(0, i + j))];
      for (let d = 0; d < D; d++) acc[d] += p[d];
      c++;
    }
    out.push(acc.map((v) => v / c));
  }
  out.push(pts[pts.length - 1]);
  return out;
}

/** Упрощение ломаной (Рамер — Дуглас — Пекер) без рекурсии. */
export function rdp(pts, eps) {
  if (pts.length < 3) return pts;
  const keep = new Uint8Array(pts.length);
  keep[0] = keep[pts.length - 1] = 1;
  const stack = [[0, pts.length - 1]];
  while (stack.length) {
    const [a, b] = stack.pop();
    const ax = pts[a][0], ay = pts[a][1], ex = pts[b][0] - ax, ey = pts[b][1] - ay, el = Math.hypot(ex, ey) || 1e-9;
    let md = -1, mi = -1;
    for (let i = a + 1; i < b; i++) {
      const d = Math.abs((pts[i][0] - ax) * ey - (pts[i][1] - ay) * ex) / el;
      if (d > md) { md = d; mi = i; }
    }
    if (md > eps) { keep[mi] = 1; stack.push([a, mi], [mi, b]); }
  }
  return pts.filter((_, i) => keep[i]);
}

/*
 * Область из мазков: прежняя область и кружки мазка (радиус кисти) — на сетку мира, sub — мазок стирает.
 * Контур закрашенного — квадраты-марши по сетке; берётся самый большой контур, упрощается до 200 точек.
 */
export function paintArea(prev, pts, r, sub) {
  let x0 = Infinity, y0 = Infinity, x1 = -Infinity, y1 = -Infinity;
  const grow = (x, y, m) => { x0 = Math.min(x0, x - m); y0 = Math.min(y0, y - m); x1 = Math.max(x1, x + m); y1 = Math.max(y1, y + m); };
  if (prev) for (const [x, y] of prev) grow(x, y, 0);
  if (!sub) for (const [x, y] of pts) grow(x, y, r);
  if (!(x1 > x0)) return null;
  const cell = Math.max(r / 3, Math.max(x1 - x0, y1 - y0) / 300);
  const ox = x0 - cell * 2, oy = y0 - cell * 2;
  const cols = Math.ceil((x1 - x0) / cell) + 5, rows = Math.ceil((y1 - y0) / cell) + 5;
  const g = new Uint8Array(cols * rows);
  if (prev && prev.length >= 3)
    for (let j = 1; j < rows - 1; j++) for (let i = 1; i < cols - 1; i++) if (inPoly(prev, ox + i * cell, oy + j * cell)) g[j * cols + i] = 1;
  const rc = Math.ceil(r / cell);
  for (const [x, y] of pts) {
    const ci = Math.round((x - ox) / cell), cj = Math.round((y - oy) / cell);
    for (let j = Math.max(1, cj - rc); j <= Math.min(rows - 2, cj + rc); j++)
      for (let i = Math.max(1, ci - rc); i <= Math.min(cols - 2, ci + rc); i++)
        if (Math.hypot(ox + i * cell - x, oy + j * cell - y) <= r) g[j * cols + i] = sub ? 0 : 1;
  }
  const loops = marchLoops(g, cols, rows);
  let best = null, bestA = 0;
  for (const L of loops) {
    let a = 0;
    for (let k = 0, m = L.length - 1; k < L.length; m = k++) a += L[m][0] * L[k][1] - L[k][0] * L[m][1];
    if (Math.abs(a) > bestA) { bestA = Math.abs(a); best = L; }
  }
  if (!best || best.length < 3) return null;
  // контур по сетке ступенчатый: два прохода сглаживания по кругу, потом упрощение
  let out = best.map(([i, j]) => [ox + i * cell, oy + j * cell]), eps = cell * 0.35;
  for (let pass = 0; pass < 2; pass++) out = out.map((q, k) => { const a = out[(k + out.length - 1) % out.length], c = out[(k + 1) % out.length]; return [(a[0] + 2 * q[0] + c[0]) / 4, (a[1] + 2 * q[1] + c[1]) / 4]; });
  let simple = rdp(out, eps);
  while (simple.length > 200) { eps *= 1.4; simple = rdp(out, eps); }
  return simple.length >= 3 ? simple.map((q) => q.map(r1)) : null;
}

/** Контуры закрашенных клеток: середины рёбер сетки, у каждой ровно два соседа — обход по кругу. */
export function marchLoops(g, cols, rows) {
  const H = (i, j) => (j * cols + i) * 2, V = (i, j) => (j * cols + i) * 2 + 1;
  const adj = new Map();
  const link = (a, b) => {
    (adj.get(a) || adj.set(a, []).get(a)).push(b);
    (adj.get(b) || adj.set(b, []).get(b)).push(a);
  };
  for (let j = 0; j < rows - 1; j++)
    for (let i = 0; i < cols - 1; i++) {
      const c = (g[j * cols + i] << 3) | (g[j * cols + i + 1] << 2) | (g[(j + 1) * cols + i + 1] << 1) | g[(j + 1) * cols + i];
      if (!c || c === 15) continue;
      const T = H(i, j), B = H(i, j + 1), L = V(i, j), R = V(i + 1, j);
      const segs = { 1: [[L, B]], 2: [[B, R]], 3: [[L, R]], 4: [[T, R]], 5: [[T, R], [L, B]], 6: [[T, B]], 7: [[T, L]], 8: [[T, L]],
        9: [[T, B]], 10: [[T, L], [B, R]], 11: [[T, R]], 12: [[L, R]], 13: [[B, R]], 14: [[L, B]] }[c];
      for (const [a, b] of segs) link(a, b);
    }
  const at = (id) => { const e = id >> 1, i = e % cols, j = (e / cols) | 0; return id & 1 ? [i, j + 0.5] : [i + 0.5, j]; };
  const seen = new Set(), loops = [];
  for (const startId of adj.keys()) {
    if (seen.has(startId)) continue;
    const loop = [];
    let prev = -1, cur = startId;
    while (cur !== undefined && !seen.has(cur)) {
      seen.add(cur);
      loop.push(at(cur));
      const nb = adj.get(cur);
      const next = nb[0] !== prev ? nb[0] : nb[1];
      prev = cur;
      cur = next;
    }
    if (loop.length >= 3) loops.push(loop);
  }
  return loops;
}

/** Кольцо-энсо радиуса R вокруг (cx, cy): незамкнутое, как у нового объекта. */
export function ring(cx, cy, R) {
  const pts = [];
  for (let i = 0; i <= 40; i++) {
    const a = -1.3 + (i / 40) * Math.PI * 1.8;
    pts.push([r1(cx + Math.cos(a) * R), r1(cy + Math.sin(a) * R)]);
  }
  return pts;
}

/** Прямоугольник w×h с центром (cx, cy): контур рамки (замкнутая ломаная) и область чуть внутри. */
export function rect(cx, cy, w, h, inset = 0) {
  const x0 = cx - w / 2 + inset, y0 = cy - h / 2 + inset, x1 = cx + w / 2 - inset, y1 = cy + h / 2 - inset;
  return [[x0, y0], [x1, y0], [x1, y1], [x0, y1]].map((p) => p.map(r1));
}

/** Круг из n точек (область портала). */
export function circle(cx, cy, R, n = 32) {
  const pts = [];
  for (let i = 0; i < n; i++) { const a = (i / n) * Math.PI * 2; pts.push([r1(cx + Math.cos(a) * R), r1(cy + Math.sin(a) * R)]); }
  return pts;
}
